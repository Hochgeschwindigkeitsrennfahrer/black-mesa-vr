# Black Mesa VR. Read once at launch. Quit and restart after edits.
# Player settings only. Other behavior uses the built-in defaults.
# Settings in the in-VR settings panel (hold Triangle/Y) are not listed here:
# the panel writes a line only for a setting you changed from its default and
# removes it again when you set it back, so improved defaults reach you in
# updates. Explanations of those settings below are kept for reference.

# Multiplier on the runtime's recommended per-eye size. 1.0 = recommended.
# 0.75 if the GPU is bound. Above 1.0 is extra supersampling (sharper, less
# shimmer, costs GPU). With GPU headroom try 1.2-1.4 together with
# OpenXRHelperDownsample=box below (PSVR2: 1.39 = 4016x4096 per eye).
RenderScale=1.0

# Largest eye side in pixels (2048-8192, default 4096). Above it both axes
# shrink by one factor, so the eye keeps its shape. MulticoreMode caps the
# height at 5456. Going far above 4096 can run the 32-bit game out of memory.
MaxEyeSize=4096

# box: with RenderScale above 1.0, the OpenXR helper averages the big eye
# image down to the headset's recommended size itself (cleaner edges, less
# shimmer). Empty = the big image goes to the runtime, which scales it.
OpenXRHelperDownsample=

# Water and mirror-floor reflections as a fraction of the eye size (default
# 0.5). 0 = the engine's fixed 1024x1024, which shimmers at floor and wall
# edges at headset resolution. Costs GPU in rooms with big reflective floors.
ReflectionRtScale=0.5

# The screen copy shiny floors and wired glass sample (_rt_PowerOfTwoFB) as a
# fraction of the eye size. At the engine's 1024x1024 the floor next to walls
# and other floor types shows a blown-up copy of that edge: jagged junctions
# that RenderScale and FXAA cannot fix. Only a copy, so much cheaper than
# ReflectionRtScale. 0.5 is recommended in every case, also with
# ReflectionRtScale=0. -1 (or unset) = same as ReflectionRtScale, 0 = 1024x1024.
RefractCopyRtScale=0.5

# Cubemap reflection strength (0..1, 1 = as the game; default 0.4 / 0.8). Lower values
# calm shiny surfaces that shimmer in the headset, without mat_specular 0
# losing all shine. World = walls, floors, glass; Models = props and NPCs.
# Your own weapons and hands are never changed. Also in the VR panel and as
# bmvr_envmap_scale_world / bmvr_envmap_scale_models (live).

# true: the world renders at the eye size.
# false: the world stays at the desktop window and is stretched into the eyes.
WorldRenderAtEyeSize=true

# Right-stick turn. Degrees per millisecond at full deflection.

# Cutscene cameras (the fall and drag in the soldier ambush, the tram) take the
# view with them; the head turns relative to them. Needs strong VR legs: you
# are tipped onto your back and dragged. false = head only (the view stays
# level and the scene plays around you). Also in the VR settings panel.

# Extra nudge on top of the built-in weapon poses (Source units / degrees).
ViewmodelPosOffsetX=0.0
ViewmodelPosOffsetY=0.0
ViewmodelPosOffsetZ=0.0
ViewmodelAngOffsetX=0.0
ViewmodelAngOffsetY=0.0
ViewmodelAngOffsetZ=0.0
ViewmodelScale=0.68

# Grip tilt, degrees. WMR uses ControllerPitchTilt. Touch and Index use their own.
ControllerPitchTilt=-35.0
ControllerPitchTiltTouch=0
ControllerPitchTiltIndex=0
ControllerPitchTiltVive=-45.0
# Player preference on top of the per-controller tilt above, in degrees
# (negative = gun points lower). Also in the VR settings panel (bmvr_gun_pitch).
# Hand models rolled thumb-up around the controller, in degrees, to match a
# hand around a grip instead of lying flat (90 = palms facing each other).
# Mirrored per hand, all controllers (Index included); not for a hand on a
# weapon. Also in the VR settings panel (bmvr_hand_roll).
# Support hand on the gun, game units along the gun: right, up, toward the muzzle
# (bmvr_hand_gun_right, _up, _forward). Also in the VR settings panel.
# While the G-Man is in front of you, grip extends only the middle finger.
# Touch aim origin sits ahead of the grip. Pulls the gun back along -forward.
ViewmodelPosOffsetXTouch=5.5

# openxr = 64-bit helper. openvr = SteamVR compositor.
VRRuntimeBackend=openxr
OpenXRHelper=true

# HUD quad, metres, relative to the headset.
HudDistance=1.05
HudSize=0.70

IPDScale=1.0
HeightOffset=0.0

# Scene MSAA. 0 / 2 / 4 / 8 / 16. Restart after editing. Not the video menu.
# 0 = off (recommended): at 2 or more the mod switches the game's FXAA off,
# and this MSAA only reaches the lighting buffers, so edges get little or no
# anti-aliasing. With 0, pick FXAA in the video options.
Msaa=0

# Render each eye's own (asymmetric) HMD frustum instead of one symmetric
# frustum that covers both eyes. On asymmetric HMDs (PSVR2: 61.5 deg out,
# 43.5 deg in) the symmetric frustum wastes ~46 % of the eye pixels, so the
# same RenderScale gets much sharper. OpenXR helper submits the runtime's
# per-eye FOV. Restart required. If something looks wrong on your headset, try false.
OffCenterEyes=true

Haptics=true
HideCrosshair=true
# Aim marker as a red laser dot instead of the amber ticks. Follows the shot
# (tilt and GunPitchOffset included); guns only. Also in the VR settings panel.
DisableViewBob=true
LeftHanded=false

# Crossbow zoom as a fraction of the headset field of view. 0.28 is about 3.5x.
ScopeZoomFovScale=0.28
# Eye the scoped crossbow bolt starts from, so it lands on that eye's reticle: right, left or center.
ScopeEye=right
# Zoomed view follows the gun's aim instead of the head: aim by pointing the crossbow (a
# two-hand grip on the front of the crossbow steadies it). The whole headset view moves with
# the hand, so it can be less comfortable. ScopeGunSmoothSec smooths hand tremor (seconds,
# 0 = raw, higher = steadier but lags). Both also in the VR settings panel (CROSSBOW SCOPE).
# Raise to zoom: hold the crossbow with both hands and bring it to your face (gun hand within
# ScopeRaiseMeters of the headset, pointing roughly where you look) to zoom in; move it
# 0.10 m further away to zoom out. Holding alt-fire still zooms by hand. Also in the VR
# settings panel (CROSSBOW SCOPE).

# Right stick down: crouch only while held (false = toggle). Left stick click: sprint latches
# until the stick is centred again (false = hold the click).
# Right stick up/down: how far (0.3..0.95) it must go before it jumps/crouches; it also has
# to point within 30 degrees of straight up/down. TurnBlocksJumpCrouch: after the stick went
# sideways to turn, no jump/crouch until it is back near the centre.
# Two-handed guns: squeezing the support grip on the gun latches the hold (no need to
# keep squeezing); squeeze again, or take the hand away from the gun, to let go.
# Frag grenade: thrown from the hand at the hand's speed (smoothed over ~66 ms around the
# fastest moment before the trigger is let go) x 2/3 x ThrowScale, capped at 1500 units/s.
# Also in the VR settings panel (Throw strength).
# false = the game's fixed throw along the aim.
# Mounted guns (50cal/TOW): how far the barrel turns per degree the hands swing around the
# mount. 1 = the handle stays in the hand; higher = less reach to aim far down/up/sideways.
# Also in the VR settings panel (MOUNTED GUNS).
# Main menu / pause panel: distance and width in metres, height offset (+ = up), and menu
# pointer smoothing in seconds (lower = more direct). Also in the VR settings panel (MENU).
# The VR settings panel (weapon menu: the VR SETTINGS slot below the crowbar)
# changes several of these live and writes the ones you changed back here.
# They are also bmvr_* console cvars.

# Hand behind the shoulder + grip: left = quick save (hold the hand there
# ~0.25 s until a light buzz, then grip; GAME SAVED shown), right = crowbar
# (grip right away). Also in the VR settings panel (SHOULDER GESTURES).

# Hold Triangle/Y 0.6 s = VR settings panel; a tap pauses on release.
# false = Triangle/Y pauses on press; open the panel from the weapon menu's gear.

# Weapon menu: VR gear slot below the crowbar (opens the VR settings panel).
# false hides it, but only while SettingsTriangleHold=true (else the panel
# could not be opened at all).
# Weapon menu opened low (hand by the hip) tilts back to face the eye
# (max 60 degrees); false = always upright.
# The VR settings panel pauses the game while it is open.
VrSettingsPause=true

# Bloom and auto-exposure on the eyes (bmvr_eye_bloom). Also in the VR settings panel.

# Deferred lights are culled from each eye's view (bmvr_light_cull_eye). false = the main view,
# which follows the gun aim: lights behind the hand go out.
LightCullEyeView=true

# Lens flares laid out around each eye's view axis, so both eyes show the same flare
# (bmvr_flare_eye_axis). false = around the image centre, as on a flat screen.
FlareEyeAxis=true
# Lens flare streaks stay level with the horizon when you tilt your head (bmvr_flare_level).
# false = they turn with the head, as on a flat screen.
FlareLevel=true
# Both eyes use one flare layout, from the point between them (bmvr_flare_shared).
FlareSharedLayout=true
# A flat screen shows about 37 degrees up and down, so a flare leaves it and its built-in fade is done.
# A headset shows far more, and a flat layer that far out looks tilted: fade out by this angle
# off the view axis (bmvr_flare_fade_angle, 0 = no fade).

# Light glows (sprites, lamps) get their brightness from a tiny occlusion query of a few pixels.
# In the headset every small head movement changes it, differently per eye: the glows flicker
# and the eyes disagree. GlowSmooth averages each glow over both eyes and smooths it over this
# many seconds (bmvr_glow_smooth, 0 = off). A glow hidden for GlowHideMs in an eye goes dark
# at once, so it does not shine through walls (bmvr_glow_hide_ms).
GlowHideMs=100
# Strength of the layers on the light (bmvr_flare_light).
# Flare layers placed away from the light are off by default: they wash out the view in a headset.
# Strength of the large square ones: 0 off, 1 stock (bmvr_flare_haze).
# A square layer away from the light this wide, as a fraction of the eye height, counts (bmvr_flare_haze_size).
# Strength of the other layers away from the light (bmvr_flare_ghosts). All three are in the VR settings panel.

# FXAA settings, also in the VR settings panel and as bmvr_fxaa_* console commands.
# The game never passes the mat_fxaa_* cvars to the shader and runs FXAA with all values at 0.
# FxaaApply=true (default) sends these values instead. false keeps the stock behaviour.
# Sub-pixel smoothing: 0 off, 0.25 about as sharp as with FxaaApply=false, 0.75 FXAA's own default, 1 softest.
# Edge contrast needed to smooth: 0.063 most edges, 0.166 FXAA default, 0.333 fewest.
# Darks below this contrast are skipped: 0.0312 treats most, 0.0833 FXAA default.
