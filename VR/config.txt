# Black Mesa VR — RenderScale is a multiplier on SteamVR's recommended
# per-eye size. The SteamVR overlay SS / resolution slider applies live
# (eye textures recreate after ~300ms). RenderScale itself is still read
# at DLL load; restart after editing this file.

# Multiplier on SteamVR's recommended per-eye size (OpenVR
# GetRecommendedRenderTargetSize). Independent of the desktop window.
# 16-aligned, cap 4096.
#
# 1.0 = full SteamVR recommended (can be taller than the HWND).
# 0.5–0.75 if GPU bound. Above 1.0 is extra supersampling on top of SteamVR SS.
#
# false (default working path): world still draws at the desktop window
# (2560x1440) and is stretched into the SteamVR-sized eyes. Raising SS does
# not add world pixels.
# true: grow FullFrame + G-buffer to the eye size so SteamVR SS is real
# source resolution. Revert: set false and restart. A crash disables it
# for the next launch only.
# FullFrame grows on this map load (not the next one). Log should say
# worldMatch=1 / grow LITERAL, not "upscaling window" and not
# "G-buffer must not outgrow FullFrame" after spawn. Flashlight + no
# ghost world means G-buffer and FullFrame actually match.
WorldRenderAtEyeSize=true
#
# Dragging SteamVR's resolution / SS slider recreates the eye textures after a
# short settle (no game restart). World G-buffers still need a map load to grow.
# VRRenderScale=1.25
RenderScale=1.0

# Perf: false (default) skips the leftover 16:9 desktop scene render after the
# stereo pair — saves ~30% frame time. Headset flashlight uses gbmatch eyes.
# true = proper 16:9 desktop at extra cost.
DesktopLeftoverRender=false

# Stereo vis: false keeps PVS + r_occlusion + r_portalsopenall 0. true is
# the old whole-map pair (occlusion off, portals open) — tanks FPS.
ForceOpenVis=false

# Hidden-area mesh early-Z. Off — OpenXR visibility-mask draw was a black
# overlay on the world and did not improve FPS (2026-09-05). Do not retry.
HiddenAreaMesh=false

# Dummy-draw materials after a gameplay map load. Off — that path
# null-called shaderapidx9 after the first stereo pair (2026-09-05).
MaterialPredraw=false

# CPU-wait GPU after the left-eye StretchRect from the shared backbuffer.
# The stereo blit path always waits (same-image-both-eyes hazard). This key
# is kept for the log / leftover fallback copy.
StereoBlitGpuFlush=true

# OpenXR helper path only. false: device idle + publish in the same Present
# (pre-2026-09-06 pass). WMR motion smoothing can hide framedrops. true: D3D
# event fence, no WaitDeviceIdle — higher fps, but the headset feels stuttery
# because reprojection no longer gets a regular cadence. See
# docs/performance-pass.md.
OpenXrDeferredPublish=false
# Extra ms a publish slot rests after the helper moved to a newer one. Safety
# margin only: the helper now reports which frame it finished blitting and a
# slot is never rewritten before that. 0-50.
OpenXrSlotCoolingMs=4
# How many eye copies may be in flight on the GPU before Present blocks on the
# oldest one. 1 = CPU at most one frame ahead of the GPU: the pose sent with a
# frame is at most one GPU frame old, reprojection has little to correct and
# walking stays smooth in low-fps areas. 2-3 = more throughput, more latency
# and judder (unbounded run-ahead measured at 2 frames on a 90Hz headset at
# ~75fps: tearing while turning, stutter while walking). 1-3.
OpenXrMaxPending=1

# Portal 2 VR / L4D2VR right-stick look. Degrees per ms at full deflection.
